DEFAULT_SETTINGS = {"provider": "genai", "model": "", "base_url": "", "max_tokens": 16000,

def load_settings() -> dict:
    s = dict(DEFAULT_SETTINGS)
    if SETTINGS_FILE.exists():
        s.update(json.loads(SETTINGS_FILE.read_text(encoding="utf-8")))
    return s


def save_settings(s: dict):
    DATA.mkdir(parents=True, exist_ok=True)
    _write_json(SETTINGS_FILE, s)


# ===== 2. API KEY STORAGE ===========================================================
# The key is never written into this file. It is saved in data/settings.json,
# encrypted with Windows DPAPI so only your Windows account on this PC can decrypt it.
# (Copying settings.json to another PC or user won't work - re-enter the key there.)
# On non-Windows systems it falls back to base64, which is NOT encryption.

if os.name == "nt":
    import ctypes
    from ctypes import wintypes

    class _Blob(ctypes.Structure):
        _fields_ = [("cbData", wintypes.DWORD), ("pbData", ctypes.POINTER(ctypes.c_char))]

    def _dpapi(data: bytes, encrypt: bool) -> bytes:
        buf = ctypes.create_string_buffer(data, len(data))
        blob_in = _Blob(len(data), ctypes.cast(buf, ctypes.POINTER(ctypes.c_char)))
        blob_out = _Blob()
        fn = ctypes.windll.crypt32.CryptProtectData if encrypt else ctypes.windll.crypt32.CryptUnprotectData
        if not fn(ctypes.byref(blob_in), None, None, None, None, 0, ctypes.byref(blob_out)):
            raise OSError("Windows could not " + ("encrypt" if encrypt else "decrypt") + " the API key")
        try:
            return ctypes.string_at(blob_out.pbData, blob_out.cbData)
        finally:
            ctypes.windll.kernel32.LocalFree(ctypes.cast(blob_out.pbData, ctypes.c_void_p))


def encrypt_key(key: str) -> str:
    if not key:
        return ""
    if os.name == "nt":
        return "dpapi:" + base64.b64encode(_dpapi(key.encode(), True)).decode()
    return "plain:" + base64.b64encode(key.encode()).decode()


def decrypt_key(stored: str) -> str:
    if not stored:
        return ""
    scheme, _, payload = stored.partition(":")
    raw = base64.b64decode(payload)
    if scheme == "dpapi":
        try:
            return _dpapi(raw, False).decode()
        except Exception:
            return ""          # saved by another user/PC - treat as "no key set"
    return raw.decode()


def shared_api_key() -> str:
    """Fallback used only when no key is saved in Settings.

    To share the key source your other GenAI apps use, paste the body of their
    resolve_api_key(...) here (or import it) and return the key string, e.g.:
        return resolve_api_key("...")
    Return "" when there is no key. As shipped, this does nothing."""
    return ""


def get_api_key() -> str:
    return decrypt_key(load_settings().get("api_key_enc", "")) or shared_api_key()


def api_key_source() -> str:
    if decrypt_key(load_settings().get("api_key_enc", "")):
        return "saved in Settings"
    return "from shared_api_key()" if shared_api_key() else ""


# ===== 3. MODEL PROVIDERS ===========================================================
# Every provider is a function: (system: str, messages: list, settings: dict) -> str
#   messages = [{"role": "user" | "assistant", "content": "..."}, ...]
# Raise LLMError("readable message") on failure - the page shows it to you as-is.


class LLMError(Exception):
    pass


def complete(system: str, messages: list) -> str:
    s = load_settings()
    fn = PROVIDERS.get(s["provider"])
    if not fn:
        raise LLMError(f"Unknown provider {s['provider']!r} - pick one in Settings.")
    return fn(system, messages, s)


def _model(s: dict) -> str:
    return "" if s["provider"] == "mock" else (s.get("model") or "").strip()


def _chat_endpoint(s: dict) -> str:
    base = (s.get("base_url") or "").strip().rstrip("/")
    if not base:
        return ""
    return base if base.endswith(CHAT_PATH) else base + CHAT_PATH


def _genai_complete(system: str, messages: list, s: dict) -> str:
    """Calls the GenAI API using the common "chat completions" JSON shape:
        POST <base URL>/chat/completions   Authorization: Bearer <API key>
        {"model": ..., "messages": [{"role": "system", ...}, ...], "max_tokens": ...}
    and reads the reply from data["choices"][0]["message"]["content"].
    ASSUMPTION: a base URL ending in /v1 usually means this format, but confirm it
    against your other apps or the API docs; if it differs, change the three marked
    lines. Proxies in the HTTPS_PROXY environment variable are used automatically, and
    on Windows the system certificate store is trusted.
    """
    url, key = _chat_endpoint(s), get_api_key()
    if not url:
        raise LLMError("Set the Base URL in Settings.")
    if not _model(s):
        raise LLMError("Set the Model in Settings.")
    if not key:
        raise LLMError("No API key - add it in Settings.")
    body = {                                                       # <-- 1. request body
        "model": _model(s),
        "messages": [{"role": "system", "content": system}] + messages,
        "max_tokens": int(s.get("max_tokens") or 4000),
    }
    headers = {"Content-Type": "application/json",
               "Authorization": f"Bearer {key}"}                  # <-- 2. auth header
    req = urllib.request.Request(url, data=json.dumps(body).encode(), headers=headers)
    try:
        with urllib.request.urlopen(req, timeout=300) as r:
            data = json.load(r)
    except urllib.error.HTTPError as e:
        detail = e.read().decode(errors="replace")[:500]
        raise LLMError(f"GenAI API error {e.code}: {detail}")
    except urllib.error.URLError as e:
        raise LLMError(f"Couldn't reach the GenAI API: {e.reason}")
    try:
        return data["choices"][0]["message"]["content"]           # <-- 3. reply text
    except (KeyError, IndexError, TypeError):
        raise LLMError("Unexpected response shape: " + json.dumps(data)[:500])


def _mock_complete(system: str, messages: list, s: dict) -> str:
    last = messages[-1]["content"] if messages else ""
    if "<current_layer>" in last:
        current = last.split("<current_layer>", 1)[1].split("</current_layer>", 1)[0].strip()
        return current + "\n\n## Learned from chat (mock)\n- Added by the mock provider to demo the update flow."
    heads = [l[2:] for l in system.splitlines() if l.startswith("# ")]
    return ("**Mock reply** - no model was called.\n\nSystem prompt sections:\n"
            + "\n".join(f"- {h}" for h in heads) + f"\n\nYou said: _{last[:200]}_")


PROVIDERS = {"mock": _mock_complete, "genai": _genai_complete}


# ===== 4. STORAGE ===================================================================
# data/contexts/*.json, data/agents/*.json, data/chats/*.json - plain, readable JSON.


class HttpError(Exception):
    def __init__(self, status: int, msg: str):
        super().__init__(msg)
        self.status, self.msg = status, msg


def _write_json(p: Path, obj):
    tmp = p.with_suffix(".tmp")
    tmp.write_text(json.dumps(obj, indent=2, ensure_ascii=False), encoding="utf-8")
    tmp.replace(p)


def _dir(kind: str) -> Path:
    d = DATA / kind
    d.mkdir(parents=True, exist_ok=True)
    return d


def _path(kind: str, item_id: str) -> Path:
    if not ID_RE.match(item_id or ""):
        raise HttpError(404, "Not found")
    return _dir(kind) / f"{item_id}.json"


def load(kind: str, item_id: str) -> dict:
    p = _path(kind, item_id)
    if not p.exists():
        raise HttpError(404, "Not found")
    return json.loads(p.read_text(encoding="utf-8"))


def save(kind: str, item: dict) -> dict:
    _write_json(_path(kind, item["id"]), item)
    return item


def load_all(kind: str) -> list:
    items = [json.loads(p.read_text(encoding="utf-8")) for p in _dir(kind).glob("*.json")]
    return sorted(items, key=lambda i: i.get("updated", 0), reverse=True)


def new_id() -> str:
    return uuid.uuid4().hex[:12]


def new_layer(kind: str, name: str, content: str, description: str = "", history=None) -> dict:
    now = time.time()
    return save(kind, {"id": new_id(), "name": (name or "").strip() or "Untitled",
                       "description": (description or "").strip(), "content": content or "",
                       "created": now, "updated": now, "history": (history or [])[-HISTORY_LIMIT:]})


def layer_summary(l: dict) -> dict:
    out = {k: l.get(k) for k in ("id", "name", "description", "content", "updated")}
    out["versions"] = len(l.get("history", []))
    return out


def chat_summary(c: dict) -> dict:
    out = {k: c.get(k) for k in ("id", "title", "context_id", "agent_ids", "updated")}
    out["count"] = len(c.get("messages", []))
    return out


# ===== 5. PROMPTS ===================================================================

def build_system_prompt(chat: dict) -> str:
    parts = ["You are an assistant supporting the user in their job. The sections below "
             "are standing context about who they are and how you should help. Use them "
             "without asking the user to repeat information already given here."]
    if chat.get("context_id"):
        try:
            ctx = load("contexts", chat["context_id"])
            parts.append(f"# Job context: {ctx['name']}\n\n{ctx['content'].strip()}")
        except HttpError:
            pass
    agents = []
    for aid in chat.get("agent_ids", []):
        try:
            agents.append(load("agents", aid))
        except HttpError:
            pass
    if len(agents) > 1:
        parts.append("Several specialist roles are active below. Draw on whichever fit "
                     "the question, and combine them when a question spans roles.")
    for a in agents:
        parts.append(f"# Agent role: {a['name']}\n\n{a['content'].strip()}")
    return "\n\n".join(parts)


UPDATE_SYSTEM = """You maintain a reusable instruction layer that is injected as a system \
prompt into future, unrelated conversations. Your job is to revise it using what was \
learned in a conversation.

Rules:
- Keep everything in the current layer that is still accurate. Preserve its structure.
- Add only durable, reusable knowledge: facts about the user's role, organisation, \
standing preferences, recurring procedures, terminology, formats they want, and \
corrections they made to how the assistant behaves.
- Do not add one-off details that only matter to this particular task.
- If the conversation contradicts the layer, update the layer to match the conversation.
- If the user gave an explicit instruction about what to capture, follow it.
- Keep it concise; merge duplicates rather than appending near-copies.
- Output ONLY the complete revised layer in Markdown. No preamble, no explanation."""


def build_update_request(layer: dict, kind: str, chat: dict, instruction: str) -> list:
    label = "job context (who the user is)" if kind == "contexts" else f"agent role '{layer['name']}'"
    transcript = "\n\n".join(f"[{m['role'].upper()}]\n{m['content']}" for m in chat["messages"])
    return [{"role": "user", "content":
             f"The layer being revised is the {label}.\n\n"
             f"<current_layer>\n{layer['content']}\n</current_layer>\n\n"
             f"<conversation>\n{transcript}\n</conversation>\n\n"
             f"<user_instruction>\n{instruction.strip() or '(none - use your judgement)'}\n"
             f"</user_instruction>"}]


# ===== 6. HTTP API ==================================================================
# @route(method, regex) registers a handler. Handlers get the URL groups plus
# body (parsed JSON) and query (dict of lists), and return a dict/list (sent as JSON)
# or a Raw(...) for pages and file downloads.

ROUTES = []


def route(method, pattern):
    def deco(fn):
        ROUTES.append((method, re.compile("^" + pattern + "$"), fn))
        return fn
    return deco


class Raw:
    def __init__(self, data: bytes, ctype: str, filename: str = None):
        self.data, self.ctype, self.filename = data, ctype, filename


K = "(contexts|agents)"
I = "([a-f0-9]{12})"

@route("GET", "/api/settings")
def get_settings(body, query):
    s, key = load_settings(), get_api_key()
    return {"provider": s["provider"], "model": s.get("model", ""), "base_url": s.get("base_url", ""),
            "max_tokens": s.get("max_tokens", 16000), "key_set": bool(key),
            "key_hint": ("..." + key[-4:]) if len(key) > 8 else ("set" if key else ""),
            "key_source": api_key_source(),
            "key_storage": "encrypted for your Windows account" if os.name == "nt" else "NOT encrypted (non-Windows)",
            "data_dir": str(DATA), "endpoint": _chat_endpoint(s)}


@route("POST", "/api/settings")
def set_settings(body, query):
    s = load_settings()
    if body.get("provider") in PROVIDERS:
        s["provider"] = body["provider"]
    for f in ("model", "base_url"):
        if f in body:
            s[f] = str(body[f]).strip()
    if "max_tokens" in body:
        try:
            s["max_tokens"] = max(256, int(body["max_tokens"]))
        except (TypeError, ValueError):
            raise HttpError(400, "Max tokens must be a number.")
    if body.get("clear_key"):
        s["api_key_enc"] = ""
    elif body.get("api_key"):                     # blank means "keep the saved key"
        s["api_key_enc"] = encrypt_key(body["api_key"].strip())
    save_settings(s)
    return get_settings({}, {})

@route("PUT", f"/api/{K}/{I}")
def update_layer(kind, item_id, body, query):
    layer = load(kind, item_id)
    if "content" in body and body["content"] != layer["content"]:
        layer["history"].append({"content": layer["content"], "saved": layer["updated"],
                                 "note": body.get("note", "")})
        layer["history"] = layer["history"][-HISTORY_LIMIT:]
        layer["content"] = body["content"]
    for f in ("name", "description"):
        if f in body:
            layer[f] = str(body[f]).strip()
    layer["updated"] = time.time()
    return layer_summary(save(kind, layer))

@route("POST", "/api/import")
def import_files(body, query):
    """body = {kind, files: [{name, text}]}. .md/.txt -> one layer each (a leading
    "# Title" names it); .json -> an export bundle. Always adds, never overwrites."""
    kind = body.get("kind") if body.get("kind") in KINDS else "contexts"
    created = []
    for f in body.get("files", []):
        name, text = f.get("name", "imported"), f.get("text", "")
        if name.lower().endswith(".json"):
            try:
                bundle = json.loads(text)
            except ValueError:
                raise HttpError(400, f"{name} is not valid JSON.")
            id_map = {}                     # keep imported chats pointing at imported layers
            for k in KINDS:
                for item in bundle.get(k, []):
                    new = new_layer(k, item.get("name"), item.get("content"),
                                    item.get("description"), item.get("history"))
                    id_map[item.get("id")] = new["id"]
                    created.append(new["name"])
            for chat in bundle.get("chats", []):
                chat = dict(chat, id=new_id(), context_id=id_map.get(chat.get("context_id")),
                            agent_ids=[id_map[a] for a in chat.get("agent_ids", []) if a in id_map])
                save("chats", chat)
                created.append(chat.get("title", "chat"))
        else:
            title = Path(name).stem
            m = re.match(r"\s*#\s+(.+)\n+", text)
            if m:
                title, text = m.group(1).strip(), text[m.end():]
            created.append(new_layer(kind, title, text.strip())["name"])
    return {"created": created}

@route("POST", f"/api/chats/{I}/messages")
def send_message(item_id, body, query):
    chat = load("chats", item_id)
    text = (body.get("content") or "").strip()
    if not text:
        raise HttpError(400, "Empty message.")
    history = [{"role": m["role"], "content": m["content"]} for m in chat["messages"]]
    reply = complete(build_system_prompt(chat), history + [{"role": "user", "content": text}])
    # Saved only after the call succeeds, so a failed send leaves no orphan message.
    chat["messages"] += [{"role": "user", "content": text, "ts": time.time()},
                         {"role": "assistant", "content": reply, "ts": time.time()}]
    if chat["title"] == "New chat":
        chat["title"] = text[:60] + ("..." if len(text) > 60 else "")
    chat["updated"] = time.time()
    return save("chats", chat)


@route("POST", f"/api/chats/{I}/propose_update")
def propose_update(item_id, body, query):
    """Returns a proposed revision; nothing is saved until the page PUTs it back."""
    chat = load("chats", item_id)
    kind = body.get("kind")
    if kind not in KINDS:
        raise HttpError(400, "Pick a layer to update.")
    layer = load(kind, body.get("id", ""))
    if not chat["messages"]:
        raise HttpError(400, "This chat has no messages to learn from yet.")
    proposed = complete(UPDATE_SYSTEM, build_update_request(layer, kind, chat, body.get("instruction", "")))
    return {"current": layer["content"], "proposed": proposed.strip(), "name": layer["name"]}


class Handler(BaseHTTPRequestHandler):
    def _send(self, status: int, data: bytes, ctype: str, filename: str = None):
        self.send_response(status)
        self.send_header("Content-Type", ctype)
        self.send_header("Content-Length", str(len(data)))
        self.send_header("Cache-Control", "no-store")
        if filename:
            self.send_header("Content-Disposition", f"attachment; filename*=UTF-8''{quote(filename)}")
        self.end_headers()
        self.wfile.write(data)

    def _json(self, obj, status=200):
        self._send(status, json.dumps(obj).encode("utf-8"), "application/json")

    def _dispatch(self, method):
        path, _, qs = self.path.partition("?")
        for m, rx, fn in ROUTES:
            match = rx.match(path) if m == method else None
            if not match:
                continue
            try:
                body = {}
                n = int(self.headers.get("Content-Length") or 0)
                if n:
                    body = json.loads(self.rfile.read(n).decode("utf-8"))
                result = fn(*match.groups(), body=body, query=parse_qs(qs))
            except HttpError as e:
                return self._json({"error": e.msg}, e.status)
            except LLMError as e:
                return self._json({"error": str(e)}, 502)
            except Exception as e:
                traceback.print_exc()
                return self._json({"error": f"Server error: {e}"}, 500)
            if isinstance(result, Raw):
                return self._send(200, result.data, result.ctype, result.filename)
            return self._json(result)
        self._json({"error": "Not found"}, 404)

    def do_GET(self): self._dispatch("GET")
    def do_POST(self): self._dispatch("POST")
    def do_PUT(self): self._dispatch("PUT")
    def do_PATCH(self): self._dispatch("PATCH")
    def do_DELETE(self): self._dispatch("DELETE")

    def log_message(self, *args):
        pass                                    # keep the console quiet

async function api(path, opts = {}) {
  const init = { method: opts.method || (opts.json !== undefined ? "POST" : "GET") };
  if (opts.json !== undefined) {
    init.headers = { "Content-Type": "application/json" };
    init.body = JSON.stringify(opts.json);
  }
  const r = await fetch(path, init);
  const data = await r.json().catch(() => ({}));
  if (!r.ok) throw new Error(data.error || `Request failed (${r.status})`);
  return data;
}
function toast(msg) {
  const t = $("#toast"); t.textContent = msg; t.hidden = false;
  clearTimeout(toast.h); toast.h = setTimeout(() => (t.hidden = true), 2600);
}
const esc = (s) => String(s).replace(/[&<>"']/g, (c) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[c]));
const when = (ts) => new Date(ts * 1000).toLocaleString([], { month: "short", day: "numeric", hour: "numeric", minute: "2-digit" });
const byId = (list, id) => list.find((x) => x.id === id);

// Minimal Markdown: code blocks, inline code, headings, lists, bold/italic, paragraphs.
function md(src) {
  const blocks = [];
  src = src.replace(/```[^\n]*\n([\s\S]*?)```/g, (_, code) => {
    blocks.push(`<pre><code>${esc(code)}</code></pre>`); return `\u0000${blocks.length - 1}\u0000`;
  });
  const inline = (s) => esc(s)
    .replace(/`([^`]+)`/g, "<code>$1</code>")
    .replace(/\*\*([^*]+)\*\*/g, "<b>$1</b>")
    .replace(/(^|[^*])\*([^*\n]+)\*/g, "$1<i>$2</i>")
    .replace(/(^|\W)_([^_\n]+)_(?=\W|$)/g, "$1<i>$2</i>");
  const out = []; let list = null, para = [];
  const flushPara = () => { if (para.length) out.push(`<p>${para.map(inline).join("<br>")}</p>`); para = []; };
  const flushList = () => { if (list) out.push(`<${list.t}>${list.items.map((i) => `<li>${inline(i)}</li>`).join("")}</${list.t}>`); list = null; };
  for (const line of src.split("\n")) {
    let m;
    if (/^\u0000\d+\u0000$/.test(line.trim())) { flushPara(); flushList(); out.push(blocks[+line.trim().slice(1, -1)]); }
    else if ((m = line.match(/^(#{1,4})\s+(.*)/))) { flushPara(); flushList(); out.push(`<h${m[1].length}>${inline(m[2])}</h${m[1].length}>`); }
    else if ((m = line.match(/^\s*[-*]\s+(.*)/)) || (m = line.match(/^\s*\d+[.)]\s+(.*)/))) {
      flushPara(); const t = /^\s*\d/.test(line) ? "ol" : "ul";
      if (!list || list.t !== t) { flushList(); list = { t, items: [] }; }
      list.items.push(m[1]);
    }
    else if (!line.trim()) { flushPara(); flushList(); }
    else { flushList(); para.push(line); }
  }
  flushPara(); flushList();
  return out.join("");
}

function renderChat() {
  const c = state.chat;
  $("#chat-title").value = c.title;
  const ctx = byId(state.contexts, c.context_id);
  const chips = [ctx ? `<span class="chip ctx" title="Job context">${esc(ctx.name)}</span>`
                     : `<span class="chip none">no job context</span>`];
  for (const id of c.agent_ids) {
    const a = byId(state.agents, id);
    chips.push(a ? `<span class="chip" title="Agent">${esc(a.name)}</span>` : `<span class="chip none">deleted agent</span>`);
  }
  if (!c.agent_ids.length) chips.push(`<span class="chip muted">no agents</span>`);
  $("#chat-layers").innerHTML = chips.join("");
  const box = $("#messages"); box.innerHTML = "";
  c.messages.forEach((m) => box.appendChild(msgEl(m.role, m.content)));

}
function msgEl(role, content, cls = "") {
  const d = document.createElement("div");
  d.className = `msg ${role} ${cls}`;
  const body = role === "assistant" && !cls ? md(content) : esc(content);
  d.innerHTML = `<div class="who">${role === "user" ? "You" : "Assistant"}</div><div class="body">${body}</div>`;
  if (role === "assistant" && !cls) {
    const a = document.createElement("div"); a.className = "actions";
    const b = document.createElement("button"); b.textContent = "Copy";
    b.onclick = () => navigator.clipboard.writeText(content).then(() => toast("Copied"));
    a.appendChild(b); d.appendChild(a);
  }
  return d;
}
async function send(text) {
  const box = $("#messages");
  box.appendChild(msgEl("user", text));
  const pending = msgEl("assistant", "Thinking…", "pending");
  box.appendChild(pending); box.scrollTop = box.scrollHeight;
  $("#btn-send").disabled = true;
  try {
    state.chat = await api(`/api/chats/${state.chat.id}/messages`, { json: { content: text } });
    renderChat(); loadChats();
  } catch (e) {
    pending.remove(); box.lastChild.remove();          // drop the optimistic user bubble
    box.appendChild(msgEl("assistant", e.message, "error"));
    $("#input").value = text;                           // give the text back to retry
  } finally { $("#btn-send").disabled = false; }
}

class Server(ThreadingHTTPServer):
    # The stdlib default (True) lets two programs share a port on Windows, so a port
    # already in use wouldn't be detected and the browser could reach the other app.
    allow_reuse_address = os.name != "nt"
    daemon_threads = True


def serve(port: int, open_browser: bool):
    seed_examples()
    server = None
    for p in range(port, port + 10):                 # step past ports already in use
        try:
            server = Server(("127.0.0.1", p), Handler)
            break
        except OSError:
            continue
    if not server:
        sys.exit(f"No free port between {port} and {port + 9}.")
    url = f"http://127.0.0.1:{server.server_port}"
    print(f"Job-Context Client {APP_VERSION} running at {url}")
    print(f"Data folder: {DATA}")
    print("Close this window (or press Ctrl+C) to stop the app.")
    if open_browser:
        threading.Timer(0.8, lambda: webbrowser.open(url)).start()
    try:
        server.serve_forever()
    except KeyboardInterrupt:
        pass
